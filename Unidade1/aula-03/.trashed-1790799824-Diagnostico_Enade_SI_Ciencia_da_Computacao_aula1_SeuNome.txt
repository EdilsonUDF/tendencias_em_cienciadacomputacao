    function corrigirQuiz() {

      let acertos = 0;

      Object.keys(gabarito).forEach((questao) => {

        const marcada =
          document.querySelector(
            `input[name="${questao}"]:checked`
          );

        const opcoes =
          document.querySelectorAll(
            `input[name="${questao}"]`
          );

        opcoes.forEach(input => {

          const label = input.closest(".option");

          label.classList.remove("correct", "wrong");

          if (input.value === gabarito[questao]) {
            label.classList.add("correct");
          }

          if (
            marcada &&
            input === marcada &&
            input.value !== gabarito[questao]
          ) {
            label.classList.add("wrong");
          }

        });

        if (marcada && marcada.value === gabarito[questao]) {
          acertos++;
        }

      });


      const score = document.getElementById("score");
      const result = document.getElementById("result");
      const title = document.getElementById("resultTitle");
      const text = document.getElementById("resultText");

      score.textContent = `${acertos}/10`;

      if (acertos >= 9) {

        title.textContent =
          "Conhecimentos bem consolidados";

        text.textContent =
          "Priorize velocidade, aprofundamento e argumentação.";

      } else if (acertos >= 7) {

        title.textContent =
          "Desempenho adequado";

        text.textContent =
          "Revise os erros específicos e busque integrar diferentes conteúdos.";

      } else if (acertos >= 5) {

        title.textContent =
          "Conhecimentos parcialmente consolidados";

        text.textContent =
          "Organize uma revisão orientada pelas áreas em que ocorreram os erros.";

      } else {

        title.textContent =
          "Lacunas importantes";

        text.textContent =
          "Monte um plano individual de estudos e priorize os fundamentos.";

      }

      result.classList.add("visible");

      result.scrollIntoView({
        behavior: "smooth",
        block: "center"
      });

    }


    function resetQuiz() {

      document
        .querySelectorAll(".option")
        .forEach(option => {
          option.classList.remove("correct", "wrong");
        });

      document
        .getElementById("result")
        .classList.remove("visible");

    }


    /* Destaque automático do item da navegação */

    const sections =
      document.querySelectorAll("main section[id]");

    const links =
      document.querySelectorAll(".nav-link");

    const observer =
      new IntersectionObserver(
        entries => {

          entries.forEach(entry => {

            if (entry.isIntersecting) {

              links.forEach(link =>
                link.classList.remove("active")
              );

              const active =
                document.querySelector(
                  `.nav-link[href="#${entry.target.id}"]`
                );

              if (active) {
                active.classList.add("active");
              }

            }

          });

        },
        {
          rootMargin: "-25% 0px -65% 0px"
        }
      );

    sections.forEach(section =>
      observer.observe(section)
    );
